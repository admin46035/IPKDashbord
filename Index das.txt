}

/* ---------- Download Helper (base64 → file di browser) ---------- */
function b64ToBlob(b64, mime){
  const bin = atob(b64), len = bin.length;
  const bytes = new Uint8Array(len);
  for (let i = 0; i < len; i++) bytes[i] = bin.charCodeAt(i);
  return new Blob([bytes], { type: mime || 'application/octet-stream' });
}
function downloadBlob(blob, name){
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = name;
  a.style.display = 'none';
  document.body.appendChild(a);
  a.click();
  setTimeout(function(){ URL.revokeObjectURL(url); a.remove(); }, 1500);
}
function formatBytes(n){
  if (!n && n !== 0) return '-';
  if (n < 1024) return n + ' B';
  const u = ['KB','MB','GB']; let i = -1;
  do { n /= 1024; i++; } while (n >= 1024 && i < u.length - 1);
  return n.toFixed(n < 10 ? 1 : 0) + ' ' + u[i];
}

/* ---------- Upload Gambar ---------- */
function readFile(file){
  return new Promise(function(res, rej){
    const r = new FileReader();
    r.onload = function(){ res(r.result); };
    r.onerror = rej;
    r.readAsDataURL(file);
  });
}
async function prepareImage(file, maxDim){
  const dataUrl = await readFile(file);
  if (file.size <= 300 * 1024){
    return { b64: dataUrl.split(',')[1], mime: file.type, name: file.name };
  }
  const img = await new Promise(function(res, rej){
    const i = new Image(); i.onload = function(){ res(i); }; i.onerror = rej; i.src = dataUrl;
  });
  const scale = Math.min(1, maxDim / Math.max(img.width, img.height));
  const w = Math.max(1, Math.round(img.width * scale));
  const h = Math.max(1, Math.round(img.height * scale));
  const c = document.createElement('canvas'); c.width = w; c.height = h;
  const ctx = c.getContext('2d');
  if (file.type !== 'image/png'){ ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, w, h); }
  ctx.drawImage(img, 0, 0, w, h);
  const png = file.type === 'image/png';
  const out = png ? c.toDataURL('image/png') : c.toDataURL('image/jpeg', 0.85);
  return { b64: out.split(',')[1], mime: png ? 'image/png' : 'image/jpeg',
           name: file.name.replace(/\.\w+$/, '') + (png ? '.png' : '.jpg') };
}
async function handleUpload(evt, targetInput, previewId, maxDim){
  const file = evt.target.files[0];
  evt.target.value = '';
  if (!file) return;
  if (!/^image\//.test(file.type)){ toast('File harus berupa gambar.', 'error'); return; }
  if (!state.token){ toast('Login diperlukan untuk upload.', 'error'); return; }
  toast('Mengunggah gambar…');
  try {

function currentScriptWebsite(){ return $('scriptWebsite').value || ''; }

function populateScriptWebsites(){
  const sel = $('scriptWebsite');
  const prev = currentScriptWebsite();
  if (!state.links.length){
    sel.innerHTML = '<option value="">— Belum ada website —</option>';
    sel.disabled = true;
    renderScriptFiles([]);
    return;
  }
  sel.disabled = false;
  sel.innerHTML = state.links.map(function(l){
    return '<option value="' + esc(l.id) + '">' + esc(l.name) + (l.scriptId ? ' 🔗' : '') + '</option>';
  }).join('');
  if (prev && state.links.some(function(l){ return l.id === prev; })) sel.value = prev;
}

function renderScriptFiles(files){
  const list = $('scriptFileList');
  files = files || [];
  $('scriptEmpty').hidden = files.length > 0;
  list.innerHTML = files.map(function(f){
    const ext = ((f.fileName.split('.').pop() || '').toUpperCase()).slice(0, 4);
    return '<div class="file-row">' +
      '<div class="file-ico">' + esc(ext || 'FILE') + '</div>' +
      '<div class="file-info">' +
        '<div class="file-name">' + esc(f.fileName) + '</div>' +
        '<div class="file-meta">' + formatBytes(f.size) + ' · ' + esc(f.updatedAt || '-') + '</div>' +
      '</div>' +
      '<div class="admin-actions">' +
        '<button class="mini-btn" title="Unduh file" onclick="downloadOne(\'' + esc(f.fileId) + '\')">' + ICONS.download + '</button>' +
        '<button class="mini-btn danger" title="Hapus file" onclick="deleteOne(\'' + esc(f.id) + '\')">' + ICONS.trash + '</button>' +
      '</div></div>';
  }).join('');
}

async function loadScriptFiles(){
  const wid = currentScriptWebsite();
  if (!wid || !state.token){ renderScriptFiles([]); return; }
  try {
    renderScriptFiles(await api('listScriptFiles', state.token, wid));
  } catch(err){
    if (String(err.message).indexOf('UNAUTHORIZED') !== -1){
      clearSession(); toast('Sesi berakhir, silakan login ulang.', 'error');
    }
    renderScriptFiles([]);
  }
}

/* Upload manual banyak file script sekaligus */
 $('fileScripts').addEventListener('change', async function(e){
  const files = Array.prototype.slice.call(e.target.files);
  e.target.value = '';
  const wid = currentScriptWebsite();
  if (!wid){ toast('Pilih website terlebih dahulu.', 'error'); return; }
  if (!files.length) return;
  const payload = [];
  for (const f of files){
    if (f.size > 8 * 1024 * 1024){ toast('“' + f.name + '” melebihi 8 MB — dilewati.', 'error'); continue; }
    const dataUrl = await readFile(f);
    payload.push({ name: f.name, mime: f.type || 'text/plain', b64: dataUrl.split(',')[1] });
  }
  if (!payload.length) return;
  toast('Mengunggah ' + payload.length + ' file…');
  try {
    const res = await api('uploadScriptFiles', state.token, wid, payload);
    if (res && res.success){
      renderScriptFiles(res.files);
      toast('Upload berhasil ✅');
    } else toast((res && res.message) || 'Upload gagal.', 'error');
  } catch(err){ toast('Gagal: ' + err.message, 'error'); }
});
